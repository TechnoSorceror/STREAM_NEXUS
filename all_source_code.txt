import React, { useEffect, useState } from 'react';
import { getMyList, MyListItem, removeFromMyList } from '../lib/myList';
import { WatchStatus } from '../types';
import MediaCard from '../components/MediaCard';

const MyList: React.FC = () => {
  const [items, setItems] = useState<MyListItem[]>([]);
  const [filter, setFilter] = useState<WatchStatus | 'all'>('all');

  useEffect(() => {
    setItems(getMyList());
  }, []);

  const handleRemove = (e: React.MouseEvent, id: string | number, type: 'movie' | 'tv' | 'anime' | 'manga') => {
    e.preventDefault();
    e.stopPropagation();
    removeFromMyList(id, type);
    setItems(getMyList());
  };

  const filteredItems = items.filter(item => {
    if (filter === 'all') return true;
    return item.watchStatus === filter || (!item.watchStatus && filter === 'plan_to_watch'); // default if undefined
  });

  return (
    <div className="pt-8 pb-20 shell space-y-12 bg-nexus-ink text-nexus-text font-mono">
      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4 mb-6">
        <h1 className="text-3xl font-bold text-nexus-text border-l-4 border-nexus-lime pl-3">My List</h1>
        <div className="flex gap-2">
          {['all', 'watching', 'plan_to_watch', 'completed', 'on_hold', 'dropped'].map((status) => (
            <button
              key={status}
              onClick={() => setFilter(status as any)}
              className={`px-3 py-1.5 rounded-full text-xs font-bold uppercase tracking-wider transition ${
                filter === status 
                ? 'bg-nexus-lime text-nexus-text' 
                : 'bg-nexus-base text-nexus-muted hover:bg-nexus-line hover:text-nexus-text'
              }`}
            >
              {status.replace(/_/g, ' ')}
            </button>
          ))}
        </div>
      </div>
      
      {filteredItems.length === 0 ? (
        <div className="flex flex-col items-center justify-center py-20 text-nexus-muted">
          <p className="text-xl">Your list is empty.</p>
          <p className="mt-2 text-sm">Add movies and TV shows to keep track of what you want to watch.</p>
        </div>
      ) : (
        <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-6 gap-4">
          {filteredItems.map((item) => (
            <MediaCard 
              key={`${item.type}-${item.id}`} 
              item={item} 
              onRemove={(e) => handleRemove(e, item.id, item.type)}
            />
          ))}
        </div>
      )}
    </div>
  );
};

export default MyList;

              let found = false;
              if (tmdbTvResults && tmdbTvResults.length > 0) {
                  result = await tmdbService.getDetails(Number(tmdbTvResults[0].id), 'tv');
                  found = !!result;
              }
              if (!found) {
                  const tmdbMovieResults = await tmdbService.search(anilistData.title, 'movie');
                  if (tmdbMovieResults && tmdbMovieResults.length > 0) {
                      result = await tmdbService.getDetails(Number(tmdbMovieResults[0].id), 'movie');
                      found = !!result;

            setPlayState({
              isPlaying: false,

import React, { useEffect, useState } from 'react';
import { useSearchParams } from 'react-router-dom';
import { tmdbService } from '../services/tmdb';
import { MediaItem } from '../types';
import MediaCard from '../components/MediaCard';
import Loader from '../components/Loader';

const SearchPage: React.FC = () => {
  const [searchParams, setSearchParams] = useSearchParams();
  const query = searchParams.get('q') || '';
  const searchType = searchParams.get('t') || 'all'; // 'media' | 'movie' | 'tv' | 'all'
  const yearParam = searchParams.get('y') || '';
  const minRatingParam = searchParams.get('r') || '0';
  
  const [results, setResults] = useState<MediaItem[]>([]);
  const [loading, setLoading] = useState(false);

  const [yearFilter, setYearFilter] = useState(yearParam);
  const [minRatingFilter, setMinRatingFilter] = useState(Number(minRatingParam));

  useEffect(() => {
    let isActive = true;
    const performSearch = async () => {
      if (!query) return;
      setLoading(true);
      
      try {
        const promises = [];
        
        if (searchType === 'media' || searchType === 'all' || searchType === 'movie') {
          promises.push(tmdbService.search(query, 'movie'));
        }

        if (searchType === 'media' || searchType === 'all' || searchType === 'tv') {
          promises.push(tmdbService.search(query, 'tv'));
        }

        const responseGroups = await Promise.all(promises);
        const flatResults = responseGroups.flat();
        
        if (isActive) setResults(flatResults);
      } catch (e) {
        console.error("Search error", e);
      } finally {
        if (isActive) setLoading(false);
      }
    };

    const debounce = setTimeout(performSearch, 500);
    return () => {
      isActive = false;
      clearTimeout(debounce);
    };
  }, [query, searchType]);

  const updateFilters = (newYear: string, newRating: number) => {
    setYearFilter(newYear);
    setMinRatingFilter(newRating);
    const params = new URLSearchParams(searchParams);
    if (newYear) params.set('y', newYear);
    else params.delete('y');
    if (newRating > 0) params.set('r', newRating.toString());
    else params.delete('r');
    setSearchParams(params);
  };

  const filteredResults = results.filter(item => {
    if (minRatingFilter > 0 && item.rating < minRatingFilter) return false;
    if (yearFilter && !item.releaseDate.startsWith(yearFilter)) return false;
    return true;
  });

  return (
    <div className="pt-8 pb-20 shell font-mono bg-nexus-ink text-nexus-text space-y-8">
      <div className="flex flex-col md:flex-row justify-between items-start md:items-center mb-6 gap-4">
        <h2 className="font-sans text-[clamp(1.7rem,3vw,2.6rem)] tracking-[-0.045em] mb-6 text-nexus-text">
          Results for "{query}"
        </h2>
        
        <div className="flex gap-4 items-center bg-nexus-base p-3 rounded-lg border border-nexus-line w-full md:w-auto">
          <div className="flex flex-col">
            <label className="text-xs text-nexus-muted font-bold uppercase tracking-wider mb-1">Year</label>
            <input 
              type="number" 
              placeholder="e.g. 2023"
              value={yearFilter}
              onChange={(e) => updateFilters(e.target.value, minRatingFilter)}
              className="bg-nexus-ink text-nexus-text text-sm px-3 py-1.5 rounded outline-none border border-nexus-line focus:border-nexus-lime w-24"
            />
          </div>
          <div className="flex flex-col">
            <label className="text-xs text-nexus-muted font-bold uppercase tracking-wider mb-1">Min Rating</label>
            <select 
              value={minRatingFilter}
              onChange={(e) => updateFilters(yearFilter, Number(e.target.value))}
              className="bg-nexus-ink text-nexus-text text-sm px-3 py-1.5 rounded outline-none border border-nexus-line focus:border-nexus-lime"
            >
              <option value={0}>Any</option>
              <option value={5}>5.0+</option>
              <option value={6}>6.0+</option>
              <option value={7}>7.0+</option>
              <option value={8}>8.0+</option>
              <option value={9}>9.0+</option>
            </select>
          </div>
        </div>
      </div>
      
      <div className="mb-6">
      </div>
      
      {loading ? (
        <div className="flex justify-center py-12">
            <Loader />
        </div>
      ) : filteredResults.length > 0 ? (
        <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-6 gap-4">
            {filteredResults.map((item, idx) => {
               const key = `${item.type}-${item.id}-${idx}`;
               return <MediaCard key={key} item={item} />;
            })}
        </div>
      ) : (
        <div className="text-center text-nexus-muted py-12 bg-nexus-base/30 rounded-lg">
            No matching results found.
        </div>
      )}

      <div className="mt-12 flex flex-col items-center gap-6">
      </div>
    </div>
  );
};

export default SearchPage;

import React, { useState, useEffect } from 'react';
import { Key, ExternalLink, Check } from 'lucide-react';
import { STORAGE_KEYS, DEFAULT_TMDB_API_KEY } from '../constants';

const OnboardingModal: React.FC = () => {
  const [isOpen, setIsOpen] = useState(false);
  const [apiKey, setApiKey] = useState('');

  useEffect(() => {
    const key = localStorage.getItem(STORAGE_KEYS.TMDB_API_KEY);
    // Only show modal if user hasn't set a key AND we don't have a default fallback
    if (!key && !DEFAULT_TMDB_API_KEY) {
      setIsOpen(true);
    }
  }, []);

  const handleSave = () => {
    if (apiKey.trim()) {
      localStorage.setItem(STORAGE_KEYS.TMDB_API_KEY, apiKey.trim());
      setIsOpen(false);
      window.location.reload(); // Reload to initialize services
    }
  };

  if (!isOpen) return null;

  return (
    <div className="fixed inset-0 z-[60] flex items-center justify-center bg-black/80 backdrop-blur-sm p-4 animate-in fade-in duration-300">
      <div className="bg-nexus-base border border-nexus-line rounded-xl shadow-2xl max-w-md w-full p-6 relative">
        <div className="text-center mb-6">
          <div className="inline-flex items-center justify-center w-12 h-12 rounded-full bg-nexus-line mb-4 text-nexus-lime">
            <Key size={24} />
          </div>
          <h2 className="text-xl font-bold text-nexus-text mb-2">Setup Required</h2>
          <p className="text-nexus-muted text-sm">
            To enable movie and TV show discovery, StreamNexus needs a TMDB API Key.

import { Star, X } from 'lucide-react';

import { Play } from 'lucide-react';

interface MediaCardProps {
  item: MediaItem & { progress?: { season?: number, episode?: number, chapter?: number } };
  index?: number;
  onRemove?: (e: React.MouseEvent) => void;
}

const MediaCard: React.FC<MediaCardProps> = ({ item, index = 0, onRemove }) => {
  const navigate = useNavigate();

  const getPath = () => {
    return `/${item.type}/${item.id}`;
  };

  const getStatusText = () => {
    if (item.progress) {
      if (item.type === 'tv' || item.type === 'anime') {
        return `S${item.progress.season || 1} E${item.progress.episode || 1}`;
      }
      if (item.type === 'manga') {
        return `Chapter ${item.progress.chapter || 1}`;
      }
      return 'Continue';
    }
    return item.releaseDate ? item.releaseDate.substring(0, 4) : '';
  };

  return (
    <Link to={getPath()} className="block h-full group">
      <div className="aspect-[2/3] overflow-hidden rounded-[8px] bg-nexus-raised border border-nexus-line relative">
        <img 
          src={item.poster} 
          alt={item.title} 
          loading="lazy"
          className="w-full h-full object-cover transition-all duration-500 ease-[cubic-bezier(0.16,1,0.3,1)] group-hover:scale-105 group-hover:saturate-[1.12]"
        />
        
        {/* Rank/Rating Badge */}
        <span className="absolute left-2 top-2 bg-nexus-ink border border-nexus-line rounded-[4px] px-[7px] py-[3px] text-nexus-lime text-[0.65rem] font-mono z-10 shadow-sm">
           #{String(index + 1).padStart(2, '0')} · {item.rating ? item.rating.toFixed(1) : 'NR'}
        </span>
        
        {/* Remove button (if continue watching) */}
        {onRemove && (
          <button 
             onClick={(e) => { e.preventDefault(); e.stopPropagation(); onRemove(e); }}
             className="absolute right-2 top-2 bg-nexus-ink/80 hover:bg-nexus-lime border border-nexus-line hover:border-nexus-lime rounded-[4px] w-6 h-6 flex items-center justify-center text-nexus-text text-xs z-10 backdrop-blur-sm transition-colors"
             aria-label="Remove from Continue Watching"
          >
             ×
          </button>
        )}

        {/* Play Button */}
        <div className="absolute right-[9px] bottom-[9px] w-[40px] h-[40px] rounded-full bg-nexus-lime text-nexus-ink grid place-items-center opacity-0 translate-y-2 transition-all duration-220 ease-[cubic-bezier(0.16,1,0.3,1)] group-hover:opacity-100 group-hover:translate-y-0 z-10 shadow-[0_4px_10px_rgba(0,0,0,0.5)]">
